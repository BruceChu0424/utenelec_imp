- 方法：每表 `BEGIN READ ONLY; SET LOCAL statement_timeout='900s'`；先按 `legacy_id IS NULL [AND legacy_import_run_id IS NULL]` 过滤出"无自身键"行，再对其调用权威函数 `fn_imported_business_document_row('<table>', to_jsonb(x))` 统计继承导入行；同时核对守卫触发器挂载与键列存在性。探针 SQL 见本文件末尾。

## 探针 SQL（复现用）

```sql
BEGIN READ ONLY;
SET LOCAL statement_timeout = '900s';
DO $$
DECLARE t text; has_legacy boolean; has_run boolean; guarded boolean;
        total bigint; keyless bigint; residual bigint; started timestamptz; key_pred text;
BEGIN
  FOREACH t IN ARRAY ARRAY['plan_order_item_links','procurement_order_approval_cases','procurement_order_qty_change_logs',
    'production_daily_report_items','production_daily_reports','production_plan_items','production_plans',
    'purchase_order_items','purchase_orders','purchase_receipt_items','purchase_receipts',
    'purchase_request_items','purchase_requests','sales_order_items','sales_orders','sales_quotes',
    'sales_shipment_items','sales_shipments','stock_document_items','stock_documents',
    'subcontract_application_items','subcontract_applications','subcontract_material_issue_items',
    'subcontract_material_issues','subcontract_order_items','subcontract_orders',
    'subcontract_receipt_items','subcontract_receipts','subcontract_return_items','subcontract_returns',
    'subcontract_waste_items','subcontract_wastes']
  LOOP
    IF to_regclass('public.'||t) IS NULL THEN RAISE NOTICE '% | MISSING_TABLE', t; CONTINUE; END IF;
    SELECT EXISTS(SELECT 1 FROM information_schema.columns WHERE table_schema='public' AND table_name=t AND column_name='legacy_id') INTO has_legacy;
    SELECT EXISTS(SELECT 1 FROM information_schema.columns WHERE table_schema='public' AND table_name=t AND column_name='legacy_import_run_id') INTO has_run;
    SELECT EXISTS(SELECT 1 FROM pg_trigger WHERE tgrelid=to_regclass('public.'||t) AND tgname='trg_00_imported_business_archive_only' AND NOT tgisinternal) INTO guarded;
    key_pred := 'TRUE';
    IF has_legacy THEN key_pred := key_pred || ' AND x.legacy_id IS NULL'; END IF;
    IF has_run THEN key_pred := key_pred || ' AND x.legacy_import_run_id IS NULL'; END IF;
    started := clock_timestamp();
    EXECUTE format('SELECT count(*) FROM public.%I', t) INTO total;
    EXECUTE format('SELECT count(*) FROM public.%I x WHERE %s', t, key_pred) INTO keyless;
    EXECUTE format('SELECT count(*) FROM public.%I x WHERE %s AND fn_imported_business_document_row(%L, to_jsonb(x))', t, key_pred, t) INTO residual;
    RAISE NOTICE '% | guarded=% | legacy_col=% | run_col=% | total=% | no_own_key=% | inherited_imported=% | %ms',
      t, guarded, has_legacy, has_run, total, keyless, residual, round(extract(epoch FROM clock_timestamp()-started)*1000);
  END LOOP;
END $$;
ROLLBACK;
```
